# 프로그램 실행에 필요한 라이브러리 설치 명령어
# CMD 또는 PowerShell에서 실행하세요

# ============================================
# 방법 1: 한 번에 설치 (권장)
# ============================================
pip install streamlit pandas numpy yfinance

# ============================================
# 방법 2: 개별 설치
# ============================================
pip install streamlit
pip install pandas
pip install numpy
pip install yfinance

# ============================================
# 방법 3: requirements.txt 사용
# ============================================
pip install -r requirements.txt

# ============================================
# 설치 확인
# ============================================
python -c "import streamlit, pandas, numpy, yfinance; print('✅ 설치 완료!')"

# ============================================
# 프로그램 실행
# ============================================
python -m streamlit run app.py
